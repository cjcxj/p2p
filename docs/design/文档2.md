```
└──────────────┬──────────────────────────────────────┘
               │
        ┌──────▼──────┐     打洞信令       ┌──────────────┐
        │   Peer A    │◄──────────────────►│  信令服务器   │
        └──────┬──────┘     (WebSocket)    │  (Lighthouse)│
               │ 加密 UDP 直连(打洞成功后)    └──────────────┘
        ┌──────▼──────┐                    ┌──────────────┐
        │   Peer B    │◄──────────────────►│  TURN 中继    │
        └─────────────┘    (对称NAT兜底)    │  (可选降级)    │
                                          └──────────────┘
```

**关键决策：**
- **信令用 WebSocket**（不是 UDP），因为要跨浏览器/移动端兼容，且信令低频
- **数据面只用加密 UDP**，直连成功即不经过任何服务器，保证极低延迟
- **加密用 Noise XK**，比 TLS 轻得多：1-RTT 握手 + 可选 0-RTT，一包即加密

## 二、技术选型

| 关注点 | 选择 | 理由 |
|---|---|---|
| 加密 | **Noise XK**（`flynn/noise`） | 1-RTT 握手、现代密码学、极轻量 |
| 数据加密 | ChaCha20-Poly1305 (AEAD) | 低延迟设备友好，无 AES-NI 也快 |
| 传输 | **UDP 数据报** | 无队头阻塞，毫秒级延迟 |
| 打洞 | STUN + 并发打洞 + TURN 兜底 | 覆盖绝大多数 NAT 类型 |
| 可靠性 | 可选层（实时数据用不可靠数据报） | 极低延迟需求下不做 TCP 式重传 |
| L3 网卡 | `water`(TUN) 或 gVisor netstack | 无 root 也能跑 netstack |
| 并发模型 | Go goroutine + 单 UDP socket | 避免多 socket 的 syscall 开销 |

## 四、核心代码骨架

### 1. 加密层 `crypto.go`（Noise XK 握手）

```go
package crypto

import (
    "crypto/rand"
    "errors"

    "github.com/flynn/noise"
    "golang.org/x/crypto/curve25519"
)

// 节点身份：长期静态密钥对（注册在信令服务器上，相当于"身份证"）
type Identity struct {
    Static noise.DHKey
    ID     [32]byte // = 公钥哈希，作为 PeerID
}

func NewIdentity() (*Identity, error) {
    priv := make([]byte, 32)
    if _, err := rand.Read(priv); err != nil {
        return nil, err
    }
    static, err := curve25519.X25519(priv, curve25519.Basepoint)
    if err != nil {
        return nil, err
    }
    return &Identity{
        Static: noise.DHKey{Private: priv, Public: static},
        ID:     [32]byte(sha256Sum(static)),
    }, nil
}

// 发起方握手：XK 模式，发起方预先知道对方的静态公钥（从信令服务器获取）
func ClientHandshake(id *Identity, peerPub []byte) (*noise.HandshakeState, error) {
    cs := noise.NewCipherSuite(noise.DH25519, noise.CipherChaChaPoly, noise.HashSHA256)
    return noise.NewHandshakeState(noise.Config{
        CipherSuite:   cs,
        Random:        rand.Reader,
        Pattern:       noise.XK,
        StaticKeypair: id.Static,
        PeerStatic:    noise.DHKey{Public: peerPub},
        Initiator:     true,
    })
}

// 数据面：握手完成后，每个 UDP 包用 AEAD 加密 + 序号防重放
func (s *SecurePacket) Encrypt(cs *noise.CipherState, nonce uint64, payload []byte) ([]byte, error) {
    if cs == nil {
        return nil, errors.New("handshake not complete")
    }
    return cs.Encrypt(nil, uint64ToBytes(nonce), payload)
}
```

### 2. 打洞核心 `holepunch.go`（状态机）

```go
package transport

// 打洞状态机
type PunchState int

const (
    Idle       PunchState = iota
    Registering            // 向信令服务器注册并获取对端
    Probing                // 双方并发互发探测包
    Established            // NAT 映射建立，可直连
    RelayFallback          // 打洞失败，降级中继
)

type HolePuncher struct {
    state     PunchState
    conn      *net.UDPConn
    signaling *SignalingClient
    selfID    crypto.PeerID
    endpoints map[crypto.PeerID][]*net.UDPAddr // 对端候选地址(本地/公网/中继)
    onPunch   func(peer crypto.PeerID, addr *net.UDPAddr)
}

// 并发打洞：这是 NAT 穿透成功的关键 —— 双方必须"同时"向对方公网地址发包
func (h *HolePuncher) Punch(peer crypto.PeerID) {
    h.state = Probing
    addrs := h.endpoints[peer] // 本地、公网、NAT猜测 三种候选
    for _, addr := range addrs {
        // 用短的、可识别的探测包，打开发通口
        probe := []byte{0x01, 'P', 'U', 'N', 'C', 'H'}
        go h.conn.WriteToUDP(probe, addr)
    }
    // 定时重发，直到收到对端回声或超时降级
    ticker := time.NewTicker(200 * time.Millisecond)
    for range ticker.C {
        if h.state == Established || h.state == RelayFallback {
            return
        }
        for _, addr := range addrs {
            h.conn.WriteToUDP(probe, addr)
        }
    }
}

// 收到任意"看起来对"的加密包即认为打洞成功
func (h *HolePuncher) OnDatagram(peer crypto.PeerID, addr *net.UDPAddr) {
    if h.state == Probing {
        h.state = Established
        h.onPunch(peer, addr)
        // 记录打通的远端地址，之后所有数据直接发这里
    }
}
```

### 3. 数据面 `datagram.go`（极低延迟要点）

```go
package transport

// 每个数据报帧结构：
// | 2B magic | 1B type | 4B seq | 24B nonce | payload(加密) | 16B tag |
// 关键：非实时的重传/排序全部放上层可选做，数据面保持"尽力而为"，
// 这样实时数据(音视频/游戏)延迟可控，可靠流由 L7 单独建。

type DatagramType uint8

const (
    DgramData   DatagramType = 0x01 // 不可靠数据，实时
    DgramReliable            = 0x02 // 可靠流分片(上层处理)
    DgramKeepAlive           = 0x03 // 保活，维持 NAT 映射
    DgramPing                = 0x04 // RTT 探测
)

// 池化，避免高频数据下的 GC 压力
var packetPool = sync.Pool{
    New: func() interface{} { return make([]byte, 0, 1500) },
}

func (t *Transport) sendDatagram(dst *net.UDPAddr, typ DatagramType, payload []byte) {
    buf := packetPool.Get().([]byte)
    defer packetPool.Put(buf)

    frame := append(buf,
        byte(magic>>8), byte(magic),
        byte(typ),
        byte(t.seq>>24), byte(t.seq>>16), byte(t.seq>>8), byte(t.seq),
    )
    // ... noise nonce + AEAD 加密 payload
    encrypted, _ := t.cipher.Encrypt(t.nonce, payload)
    frame = append(frame, encrypted...)

    // 关键低延迟优化：合并小包 / 及时 flush，减少 syscall
    t.conn.WriteToUDP(frame, dst)
    t.seq++
}
```

### 4. 网络层 `l3_tun.go`（overlay IP）

```go
package l3

import (
    "github.com/songgao/water"
    "github.com/xtaci/smux" // 或自研
)

// 用 water 创建 TUN 设备（需 root），或者用 gVisor netstack 纯用户态免 root
func NewTUN() (*water.Interface, error) {
    iface, err := water.New(water.Config{DeviceType: water.TUN})
    if err != nil {
        return nil, err
    }
    // 分配 overlay 子网 IP，例如 10.99.0.0/16
    return iface, nil
}

// 从 TUN 读 IP 包 -> 加密封装 -> 发给目标节点(按路由表)
func (l *Layer3) pump() {
    buf := make([]byte, 1500)
    for {
        n, _ := l.tun.Read(buf)
        dstIP := parseDestIP(buf[:n])
        peer := l.routing.Lookup(dstIP) // overlay 路由表
        if peer == nil {
            continue // 丢弃，不广播
        }
        l.transport.SendOverlay(peer, buf[:n]) // 走上面数据面加密
    }
}
```

## 五、极低延迟工程清单

| # | 优化 | 手段 |
|---|---|---|
| 1 | 减少 syscall | 单 UDP socket + 合并小包；批量读 `ReadBatch` |
| 2 | 零拷贝 | `sync.Pool` 复用缓冲，避免频繁 GC |
| 3 | 无队头阻塞 | 实时数据走 `DgramData`，绝不重传 |
| 4 | 打洞保活 | 每 15~30s 发 `DgramKeepAlive` 维持 NAT 映射 |
| 5 | 握手提速 | Noise 0-RTT（K 模式）重连场景免握手 |
| 6 | 丢包恢复 | 音频/游戏类用 FEC（前向纠错），不重传 |
| 7 | 双路径 | 同时测打洞直连和 relay，秒级自动切换最优路径 |
| 8 | RTT 探测 | `DgramPing` 持续测速，路由/中继动态择优 |
